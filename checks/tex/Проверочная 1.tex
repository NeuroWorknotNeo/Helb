\begin{enumerate}[leftmargin=2.2em,itemsep=1.2em]
\item Let $A$ and $B$ be $n\times n$ matrices with complex entries for which
\[
A^2+B^2=2(AB-BA).
\]
Prove that $A^2+B^2$ is a nilpotent matrix.

\item Let $(a_n)$ be a recursive sequence given by
\[
a_0=1,\qquad a_{n+1}=\frac{2a_n}{2+\sqrt{4-a_n^2}}.
\]
Prove that $(2^na_n)$ is increasing and find its limit.
\end{enumerate}
