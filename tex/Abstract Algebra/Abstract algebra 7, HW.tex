\sheettitle{Abstract algebra -- 7}{Generators and relations: infinite and matrix groups}{Homework}

\head{Theoretical minimum.} A presentation $\langle S\mid R\rangle$ is the free group $F(S)$ of
reduced words modulo the normal subgroup generated by $R$; by von Dyck's theorem every group
generated by elements satisfying $R$ is a quotient of it. Thus
$D_n=\langle s,t\mid s^2=t^2=(st)^n=e\rangle$, $\Z_2*\Z_2=\langle s,t\mid s^2=t^2=e\rangle$ is
the infinite dihedral group, and $S_n$ is the Coxeter group with
$s_i^2=(s_is_{i+1})^3=(s_is_j)^2=e$, $|i-j|\geqslant2$. Quotients and finite-index subgroups
of finitely generated groups are finitely generated; arbitrary subgroups need not be.
Ping-pong: if $a^kX_2\subseteq X_1$ and $b^kX_1\subseteq X_2$ for $k\neq0$ and
$X_1\cap X_2=\varnothing$, then $\langle a,b\rangle$ is free. An element is a product of two
involutions if and only if an involution conjugates it to its inverse; two reflections compose
to a rotation by twice the angle between the mirrors. Minkowski: an integer matrix of finite
order that is $\equiv I$ modulo an odd prime $p$ is $I$, so finite subgroups of $\GL_n(\Z)$
embed into $\GL_n(\mathbb F_p)$. Burnside: subgroups of $\GL_n(\mathbb C)$ of finite exponent
are finite.

\medskip\noindent\rule{\linewidth}{0.4pt}

\begin{enumerate}[leftmargin=2.2em,itemsep=0.6em]
\item Prove that the composition of the reflections in two lines meeting at a point $O$ at angle
$\theta$ is the rotation about $O$ by $2\theta$. Deduce that every element of the orthogonal
group $O(n)$ is a product of two involutions lying in $O(n)$, and that the reflections in the
sides of an equilateral triangle generate an infinite group. Show that in the quaternion group
$Q_8=\{\pm1,\pm i,\pm j,\pm k\}$ the element $i$ is not a product of two involutions.

\item Let the multiplicative group $F^*$ of a field $F$ be finitely generated. Must $F^*$ be
cyclic? \src{МФТИ 2024}

\item Let $G$ be a group such that every element of its commutator subgroup $G'$ (the subgroup
generated by all $xyx^{-1}y^{-1}$) has finite order. Prove that the set of all elements of $G$
of finite order is a subgroup of $G$. \src{IMS 2006}

\item Let $a,b,c$ be elements of finite order in some group. Prove that if $a^{-1}ba=b^2$,
$b^{-2}cb^2=c^2$ and $c^{-3}ac^3=a^2$, then $a=b=c=e$, where $e$ is the unit element.
\src{VJIMC 2011}

\item 
Let $p$ be an odd prime. (a) Prove Minkowski's theorem and deduce that the order of a
finite subgroup of $\GL_n(\Z)$ divides both $|\GL_n(\mathbb F_3)|$ and $|\GL_n(\mathbb F_5)|$.
(b) Prove that every finite subgroup $G$ of $\GL_n(\mathbb Q)$ is conjugate to a subgroup of
$\GL_n(\Z)$ (consider the subgroup $\sum_{g\in G}g\Z^n$ of $\mathbb Q^n$). (c) Find finite
subgroups of $\GL_2(\Z)$ of orders $8$ and $12$, and compare with the bound of (a).

\item 
Let $G\subseteq\GL_n(\mathbb C)$ be a group with $g^m=I$ for all $g\in G$. Prove that
$|G|\leqslant m^{n^3}$, assuming Burnside's theorem that an irreducible subgroup of
$\GL_n(\mathbb C)$ spans $M_n(\mathbb C)$. Show that $G$ can be infinite if (a) the exponents
are allowed to depend on $g$; (b) $\mathbb C$ is replaced by a field of characteristic $p$.

\item Let $F_2$ be the free group with two generators. Prove that the commutator subgroup of
$F_2$ is not finitely generated. \src{Высшая лига 2016}

\item Prove that the group $\SO(3)$ of proper rotations of three-dimensional space contains a
free subgroup generated by two elements. \emph{Hint:} try the rotations by $\arccos\frac35$
about two perpendicular axes, multiply a word of length $k$ in them by $5^k$ and reduce modulo
$5$. \src{МФТИ}

\item Let $p$ be a prime, and consider the group $G$ consisting of $2\times2$ matrices $\bm{y^m}{y^nf(y)}01$, where $f$ varies in $\mathbb F_p[y]$ and $m,n$ over integers. (i) Show that $G$ is finitely
generated; (ii) show that there does not exist a finite sequence of elements $g_1,\dots,g_n$
(not necessarily distinct) of $G$ such that
$G=\{g_1^{k_1}g_2^{k_2}\cdots g_n^{k_n}:k_i\in\Z\}$. \src{IMC 2025, shortlist}

\item Let $G$ be the free product $\Z_3*\Z_3=\langle x,y\mid x^3=y^3=e\rangle$. If $N_n$ denotes
the number of subgroups of index $n$ in $G$, prove that $N_n$ is odd if, and only if, $n$ is
of the form $2^k-3$. (Compare Abstract algebra -- 4.) \src{IMC 2025, shortlist}
\end{enumerate}
