\sheettitle{Abstract algebra -- 4}{Group actions, homomorphisms and automorphisms}{Homework}

\head{Theoretical minimum.} An action of $G$ on $X$ is a homomorphism $G\to\Sym(X)$; the orbit
of $x$ has $[G:G_x]$ elements, where $G_x$ is the stabiliser, so every orbit size divides $|G|$.
The number of orbits equals $\frac1{|G|}\sum_g|\Fix(g)|$ (Burnside), which is proved by counting
the pairs $(g,x)$ with $gx=x$ in two ways. Left translation gives an embedding
$G\hookrightarrow S_{|G|}$ (Cayley); the action on the cosets of a subgroup $H$ of index $n$
gives a homomorphism $G\to S_n$ with kernel the largest normal subgroup contained in $H$, whence
a subgroup of smallest prime index is normal. Conjugation gives
$\Inn(G)\cong G/\ZZ(G)\trianglelefteq\Aut(G)$. Finally $\Aut(\Z/n)\cong(\Z/n)^*$ and
$\Aut((\Z/p)^k)\cong\GL_k(\F_p)$.

\medskip\noindent\rule{\linewidth}{0.4pt}

\begin{enumerate}[leftmargin=2.2em,itemsep=0.6em]
\item Let $G$ be a finite group. For subsets $U,V,W\subseteq G$ denote by $N_{UVW}$ the number
of triples $(x,y,z)\in U\times V\times W$ with $xyz=e$. Suppose $G$ is partitioned into three
sets $A,B,C$. Prove that $N_{ABC}=N_{CBA}$. \src{IMC}

\item Suppose that a finite group $G$ is generated by two elements $g$ and $h$, where the order
of $g$ is odd. Show that every element of $G$ can be written in the form
$g^{m_1}h^{n_1}g^{m_2}h^{n_2}\cdots g^{m_r}h^{n_r}$ with $1\le r\le|G|$ and all exponents in
$\{-1,1\}$. (The IMC problem on the non-existence of an injective homomorphism
$\GL_2(\Z/p\Z)\to S_p$ is Linear algebra -- 9, homework 6.) \src{Putnam 2016, A5}

\item Let $G$ be a finite group with $n\ge2$ elements and let $p$ be the largest prime divisor
of $n$. Prove that the number of endomorphisms $f\colon G\to G$ is at most $n^{n/p}$, and find
all groups for which equality holds. \src{\ru{Румыния, 2008}}

\item For a group $G$ and $m\in\mathbb N$ let $G(m)$ be the subgroup generated by all $m$-th
powers of elements of $G$. Prove that if $G(m)$ and $G(n)$ are commutative, then so is
$G(\gcd(m,n))$. \src{IMC}

\item Prove that $\Aut(\Z/n)\cong(\Z/n)^*$ and that $\Aut((\Z/p)^k)\cong\GL_k(\F_p)$, and
compute the order of the latter. Then find a finite group $G$ with $|\Aut(G)|=1$ and prove that
there are exactly two such groups.

\item Using Burnside's lemma, count the colourings of the faces of a cube in $k$ colours up to
rotations of the cube, and the colourings of its vertices; then explain why the two answers are
polynomials in $k$ with rational but non-integral coefficients whose values are nevertheless
integers.

\item Let $G$ be a group whose operation is $*$ such that (i) $G$ is a subset of $\mathbb R^3$,
the operation being unrelated to vector addition, and (ii) for all $a,b\in G$ either
$a\times b=a*b$ or $a\times b=0$, where $\times$ is the usual cross product. Prove that
$a\times b=0$ for all $a,b\in G$. \src{Putnam 2010, A5}

\item Let a finite group $G$ act transitively on a set $X$ with $|X|>1$. Prove that some
$g\in G$ has no fixed point at all (Jordan), and show that the proportion of such $g$ is at
least $1/|X|$.

\item Prove that $\Inn(G)$ is normal in $\Aut(G)$, and that every automorphism of $S_n$ which
maps transpositions to transpositions is inner. (For $n\ne6$ every automorphism does map
transpositions to transpositions, so $\Aut(S_n)=\Inn(S_n)$; the case $n=6$ is a genuine
exception.)

\item Prove that a group with a subgroup of index $n$ has a normal subgroup of index dividing
$n!$, and deduce that a simple group of order $60$ has no subgroup of index less than $5$.
\end{enumerate}
