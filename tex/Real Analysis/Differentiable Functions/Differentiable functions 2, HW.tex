\noindent\textbf{\underline{Theoretical minimum.}}\quad Rolle, Lagrange and Cauchy mean value theorems;
the auxiliary function trick and the standard multipliers $e^{\lambda x}$, $\frac1x$, $\varphi(x)$; Darboux'
theorem; repeated application of Rolle to a function with many zeros; the mean value theorem for
integrals and its use in producing a point.

\noindent\rule{\textwidth}{0.4pt}\par\vspace{4pt}
\begin{enumerate}
\item (IMC 2025) Let $f\colon(0,\infty)\to\mathbb R$ be continuously differentiable and let $b>a>0$
satisfy $f(a)=f(b)=k$. Prove that there is $\xi\in(a,b)$ with $f(\xi)-\xi f'(\xi)=k$.

\item (RMO 1999) Let $a<b$ and let $f,g\colon[a,b]\to\mathbb R$ be differentiable with increasing
derivatives and $f'(a)>0$, $g'(a)>0$. Prove that there is $c\in[a,b]$ with
$\frac{f(b)-f(a)}{b-a}\cdot\frac{g(b)-g(a)}{b-a}=f'(c)g'(c)$.

\item (OMOUS 2022) Let $f\colon[0,1]\to\mathbb R$ be continuous with $\int_0^1f(x)\,dx=\frac\pi4$.
Prove that there is $x_0\in(0,1)$ with $\frac1{1+x_0}<f(x_0)<\frac1{2x_0}$.

\item Let $f$ be continuous on $[a,b]$ and differentiable on $(a,b)$, where $b-a\ge\pi$. Prove that
there is $x_0\in(a,b)$ with $f'(x_0)<1+f(x_0)^2$.

\item (IMC 2016) Let $f\colon[a,b]\to\mathbb R$ be continuous on $[a,b]$ and differentiable on
$(a,b)$. Suppose $f$ has infinitely many zeros but there is no $x\in(a,b)$ with $f(x)=f'(x)=0$.
\begin{enumerate}[label=\alph*)]
\item Prove that $f(a)f(b)=0$.
\item Give an example of such a function on $[0,1]$.
\end{enumerate}

\item (IMC 2005) Prove that for a three times differentiable $f\colon\mathbb R\to\mathbb R$ there is
$\xi\in(-1,1)$ with $\frac{f'''(\xi)}6=\frac{f(1)-f(-1)}2-f'(0)$.

\item Let $f\in C^1[a,b]$ with $f>0$. Prove that $\frac{f(b)-f(a)}{b-a}\le\max_{x\in[a,b]}\Bigl(f(x)+
\frac{(f'(x))^2}{4f(x)}\Bigr)$.

\item Let $f$ be continuous on $[a,b]$ and differentiable on $(a,b)$, where $0<a<b$. Prove that there
are $c_1,c_2\in(a,b)$ with $\frac{f'(c_1)}{a+b}=\frac{f'(c_2)}{2c_2}$.

\item (IMC 1995) Let $f$ be twice continuously differentiable on $(0,+\infty)$ with
$\lim_{x\to0^+}f'(x)=-\infty$ and $\lim_{x\to0^+}f''(x)=+\infty$. Show that
$\lim_{x\to0^+}\frac{f(x)}{f'(x)}=0$.

\item (IMC 2006) Find all sequences $a_0,a_1,\dots,a_n$ of reals with $n\ge1$ and $a_n\ne0$ for which
the following holds: if $f\colon\mathbb R\to\mathbb R$ is $n$ times differentiable and
$x_0<x_1<\dots<x_n$ satisfy $f(x_0)=f(x_1)=\dots=f(x_n)=0$, then there is $h\in(x_0,x_n)$ with
$a_0f(h)+a_1f'(h)+\dots+a_nf^{(n)}(h)=0$.
\end{enumerate}
