\noindent\textbf{\underline{Theoretical minimum.}}\quad Linear recurrences and the characteristic
polynomial; multiple roots and an inhomogeneous right-hand side; generating functions and partial
fractions; conjugate surds $\alpha^n+\beta^n$; the matrix form
$\binom{a_{n+1}}{a_n}=A\binom{a_n}{a_{n-1}}$; proofs of integrality by induction and by
substitution.

\noindent\rule{\textwidth}{0.4pt}\par\vspace{4pt}
\begin{enumerate}
\item Let $x_n=\frac{(x_{n-1})^2-9}8$ for $n\ge2$. Find $\lim_{n\to\infty}x_n$ for all initial
values $x_1\in\mathbb R$.

\item Let $t$ and $\varepsilon$ be real with $|\varepsilon|<1$. Prove that the equation
$x-\varepsilon\sin x=t$ has a unique real solution, and write it as the limit of an explicit
iteration.

\item (IMC 2015) Let $F(0)=0$, $F(1)=\frac32$ and $F(n)=\frac52F(n-1)-F(n-2)$ for $n\ge2$.
Determine whether $\sum_{n\ge0}\frac1{F(2^n)}$ is rational.

\item (IMC 2019) Define $a_0=1$, $a_1=2$ and $(n+3)a_{n+2}=(6n+9)a_{n+1}-na_n$ for $n\ge0$.
Prove that all terms of the sequence are integers.

\item (IMC 2003) Let $a_0=1$ and $a_{n+1}=\frac1{n+1}\sum_{k=0}^n\frac{a_k}{n-k+2}$. Find
$\lim_{n\to\infty}\sum_{k=0}^n\frac{a_k}{2^k}$.

\item (IMC 2010) Compute the sum of the series $\sum_{k=0}^\infty\frac1{(4k+1)(4k+2)(4k+3)(4k+4)}$.

\item (IMC 2010) Define $x_1=\sqrt5$ and $x_{n+1}=x_n^2-2$ for $n\ge1$. Compute
$\lim_{n\to\infty}\frac{x_1x_2\cdots x_n}{x_{n+1}}$.

\item (IMC 2008) Let $n$ be a positive integer. Prove that $2^{n-1}$ divides
$\sum_{0\le k<n/2}\binom n{2k+1}5^k$.

\item (IMC 2022) Let $p$ be a prime. A flea sits at the point $0$ of the real line. Each minute it
either stays where it is, or jumps by $1$ to the left or to the right. Let $f(p)$ be the number of
ways for the flea to be back at $0$ after $p-1$ minutes. Find $f(p)$ modulo $p$.

\item (AKHIMO 2025) Define $a_1=2$, $a_2=5$ and $a_{n+2}=f(a_{n+1},a_n)$ for $n\ge1$, where
$f(x,y)=5(x+y)+2\sqrt{6x^2+15xy+6y^2}$. Show that $a_n$ is an integer for every $n$.

\item (OMOUS 2026) Let $a_0=1$, $a_1=2$ and $a_{n+1}=\frac{3^{n-1}+a_n^2}{a_{n-1}}$ for $n\ge1$.
\begin{enumerate}[label=\alph*),leftmargin=2.64em,labelsep=0.94em]
\item Show that all $a_n$ are integers.
\item Does the series $\sum_{n\ge0}\frac1{a_n}$ converge?
\item Is the sum $\sum_{n\ge0}\frac{2^n}{a_{2^n}}$ rational?
\end{enumerate}
\end{enumerate}
