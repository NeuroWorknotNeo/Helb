\noindent\textbf{\underline{Theoretical minimum.}}\quad The lemma on a neutral fixed point
$x_{n+1}=x_n-ax_n^p+o(x_n^p)$; the Stolz--Ces\`aro theorem; telescoping in powers of $x_n$;
comparison of a sum with an integral; the asymptotics $H_n=\ln n+\gamma+O(1/n)$; the step from
``the limit is zero'' to the leading term together with an estimate of the remainder.

\noindent\rule{\textwidth}{0.4pt}\par\vspace{4pt}
\begin{enumerate}
\item Let $x_0\in[-1,1]$ and $x_{n+1}=x_n-\arcsin(\sin^3x_n)$ for $n\ge0$. Compute
$\lim_{n\to\infty}\sqrt n\,x_n$.

\item Let $f\colon\mathbb R\to\mathbb R$ be continuous and suppose the sequence
$a_n=\int_0^1f(n+x)\,dx$ converges. Prove that the sequence $b_n=\int_0^1f(nx)\,dx$ converges as
well.

\item (AKHIMO 2018) Let $x_1=2018$ and $\frac{x_{n+1}}{x_n}=2018+\frac{2018}n$ for $n\ge1$. Find
$\lim_{n\to\infty}\biggl(\frac{\log_2n}{n-\log_{2018}x_n}\biggr)$.

\item (AKHIMO 2025) Define $b_1=1$, $b_2=3$, $b_3=2$ and, for $n\ge4$, let $b_n$ be the average of
the two smaller numbers among $b_{n-1},b_{n-2},b_{n-3}$. Find $\lim_{n\to\infty}b_n$.

\item Let $c_0,c_1>0$ and $c_{n+1}=\sqrt{c_n}+\sqrt{c_{n-1}}$ for $n\ge1$. Prove that the sequence
converges and find its limit.

\item (IMC 2024) Define $x_1=2$, $x_2=4$ and $x_{n+2}=3x_{n+1}-2x_n+\frac{2^n}{x_n}$ for $n\ge1$.
Prove that $\lim_{n\to\infty}\frac{x_n}{2^n}$ exists, and estimate it.

\item (OMOUS 2025) Compute $\displaystyle\prod_{n=1}^\infty\frac{(2n)^4-1}{(2n+1)^4-1}\cdot
\frac{(n+1)^2}{n^2}$.

\item (Vienna MC 2025) Let $(a_n)$ satisfy $a_0>0$ and $a_{n+1}=\frac{a_n}{a_n^2+a_n+1}$ for all
$n$. Show that $a_n\to0$ and determine $\lim_{n\to\infty}na_n$.

\item For every real $x_1$ construct the sequence by $x_{n+1}=x_n\bigl(x_n+\frac1n\bigr)$. Prove
that there is exactly one value of $x_1$ for which $0<x_n<x_{n+1}<1$ for all $n$.

\item Let $f\colon(0,\infty)\to\mathbb R$ be continuous and suppose that
$\lim_{n\to\infty}f(nx)=0$ for every fixed $x>0$. Prove that $\lim_{x\to\infty}f(x)=0$.
\emph{(Hint: Baire's category theorem.)}
\end{enumerate}
