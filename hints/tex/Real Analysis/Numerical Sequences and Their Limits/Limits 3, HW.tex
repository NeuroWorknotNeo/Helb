\hintstitle{Limits -- 3. Asymptotics of sequences}{Homework}

\problem{1}
\hint{1} The map $g(x)=x-\arcsin(\sin^3x)$ is odd, so it suffices to take $x_0\in(0,1]$ (for $x_0=0$ the sequence is $0$). For $x\in(0,1]$: $0<\sin^3x<\sin x$, so $0<\arcsin(\sin^3x)<x$; hence $x_n$ stays in $(0,1]$, decreases, and tends to a fixed point, i.e.\ to $0$.
\hint{2} Here $\arcsin(\sin^3x)=x^3+O(x^5)$, so $x_{n+1}=x_n-x_n^3+O(x_n^5)$: Theorem 1 with $p=3$, $a=1$. Telescoping: $\dfrac1{x_{n+1}^2}-\dfrac1{x_n^2}\to2$, and Stolz gives $nx_n^2\to\frac12$.
\hint{3} Answer: $\lim\sqrt n\,x_n=\frac1{\sqrt2}$ for $x_0\in(0,1]$, $-\frac1{\sqrt2}$ for $x_0\in[-1,0)$, and $0$ for $x_0=0$.

\problem{2}
\hint{1} Substitute $t=nx$: $b_n=\frac1n\int_0^nf(t)\,dt=\frac1n\sum_{k=0}^{n-1}a_k$.
\hint{2} So $b_n$ is the Ces\`aro mean of $(a_k)$, and converges to the same limit (Stolz--Ces\`aro).

\problem{3}
\hint{1} The product equals $P_n=\dfrac{(n!)^2}{(2n)!}$, and the exponent is $e^{1999/n}-1=\dfrac{1999}n\,(1+o(1))$. So the logarithm of the expression is $1999\cdot\dfrac{\ln P_n}n\,(1+o(1))$.
\hint{2} Find $\lim\frac{\ln P_n}n$ by Stolz: it equals $\lim\bigl(\ln P_{n+1}-\ln P_n\bigr)=\lim\ln\dfrac{(n+1)^2}{(2n+1)(2n+2)}=-\ln4$. (Stirling gives more, $\ln P_n=-n\ln4+\frac12\ln(\pi n)+o(1)$, but it is not needed.)
\hint{3} The logarithm of the expression tends to $-1999\ln4$. Answer: $4^{-1999}$.

\problem{4}
\hint{1} The numerator is a geometric progression: $\frac{n^{n+1}-n}{n-1}\sim n^n$. Divide everything by $n^n$.
\hint{2} The denominator becomes $\sum_{j=0}^{n-1}\bigl(1-\frac jn\bigr)^n$. Each term tends to $e^{-j}$ and is at most $e^{-j}$ (since $1-t\le e^{-t}$), so the sum tends to $\sum_{j\ge0}e^{-j}=\frac e{e-1}$ (cut the sum at a large fixed $J$; the tail is at most $\sum_{j>J}e^{-j}$).
\hint{3} Answer: $\dfrac{e-1}e$.

\problem{5}
\hint{1} a) $g_n(x)=f(x)-x^n$ is strictly decreasing, $g_n(0)=f(0)>f(1)\ge0$ and $g_n(1)=f(1)-1<f(0)-1\le0$: one root $a_n\in(0,1)$. Since $g_{n+1}(a_n)=a_n^n-a_n^{n+1}>0$, the sequence $a_n$ increases. If $a_n\to L<1$, then $a_n^n\to0$ and $f(L)=0$, but $f(1)<f(L)$ and $f(1)\ge0$.
\hint{2} b) Put $t_n=1-a_n\to0$ and $c=-f'(1)>0$. Then $f(a_n)=ct_n(1+o(1))$ and $a_n^n=e^{-nt_n(1+o(1))}$, so taking logarithms,
\[nt_n(1+o(1))=\ln\frac1{t_n}+O(1).\]
Deduce $t_n\sim\frac{\ln n}n$ (first show $\frac1n\lesssim t_n\lesssim n^{-1/2}$, so that $\ln\frac1{t_n}\sim\ln n$).
\hint{3} $F(a_n)=\int_{a_n}^1f\sim\frac c2t_n^2\sim\frac c2\Bigl(\frac{\ln n}n\Bigr)^2$. So $\sum F(a_n)^\alpha$ behaves like $\sum\bigl(\frac{\ln n}n\bigr)^{2\alpha}$: it converges exactly for $\alpha>\frac12$ (for $\alpha\le0$ the terms do not even tend to $0$).

\problem{6}
\hint{1} Rewrite: $x_{n+2}-2x_{n+1}=(x_{n+1}-2x_n)+\frac{2^n}{x_n}$. So $d_n=x_{n+1}-2x_n$ satisfies $d_1=0$, $d_{n+1}=d_n+\frac{2^n}{x_n}$.
\hint{2} Put $y_n=x_n/2^n$. Dividing by $2^{n+1}$: $y_{n+1}-y_n=\frac{d_n}{2^{n+1}}$, and $\frac{2^n}{x_n}=\frac1{y_n}$, so $d_n=\sum_{k=1}^{n-1}\frac1{y_k}$. Hence $d_n\ge0$, $y_n$ is non-decreasing and $y_n\ge y_1=1$.
\hint{3} Sum the increments and swap the order of summation, using $\sum_{n=k+1}^{N-1}2^{-n-1}=2^{-k-1}-2^{-N}$:
\[y_N=1+\sum_{n=1}^{N-1}\frac1{2^{n+1}}\sum_{k=1}^{n-1}\frac1{y_k}=1+\sum_{k=1}^{N-2}\frac{2^{-k-1}-2^{-N}}{y_k}.\]
Since $y_k\ge1$, $y_N<1+\sum_{k\ge1}2^{-k-1}=\frac32$: the sequence is non-decreasing and bounded, so $L=\lim y_n$ exists, and letting $N\to\infty$,
\[L=1+\sum_{k\ge1}\frac1{2^{k+1}y_k}.\]
\hint{4} Now use $y_k\le L$. Already $y_k\in[1,L]$ gives $1+\frac1{2L}\le L\le\frac32$, i.e.\ $\frac{1+\sqrt3}2\le L\le\frac32$. Sharper: $y_1=y_2=1$, $y_3=\frac98$, and $y_k\in[\frac98,L]$ for $k\ge3$, so
\[\frac{11}8+\frac1{8L}\le L\le\frac{11}8+\frac19,\qquad\text{i.e.}\qquad 1.460<\frac{11+\sqrt{153}}{16}\le L\le\frac{107}{72}<1.487.\]
Numerically $L\approx1.4782$.

\problem{7}
\hint{1} Use $\frac{2x}\pi\le\sin x\le x$ on $\bigl[0,\frac\pi2\bigr]$. a) Near $0$ the integrand is $\frac{\{nx\}}{\sin x}\le\frac{nx}{\sin x}\le\frac{\pi n}2$, so it is bounded and the integral converges. b) The same inequality gives $J(n)\le I(n)\le\frac\pi2J(n)$, where, after $t=nx$,
\[J(n)=\int_0^{\pi/2}\frac{\{nx\}}x\,dx=\int_0^{n\pi/2}\frac{\{t\}}t\,dt.\]
So it suffices to show that $J(n)$ is of order $\ln n$.
\hint{2} Upper bound: $\{t\}\le\min(t,1)$, so $J(n)\le\int_0^1dt+\int_1^{n\pi/2}\frac{dt}t=1+\ln\frac{\pi n}2$. Lower bound: on each $[k,k+1]$, $\int_k^{k+1}\frac{\{t\}}t\,dt=\int_0^1\frac s{k+s}\,ds\ge\frac1{2(k+1)}$, and $[1,n]\subset\bigl[0,\frac{n\pi}2\bigr]$, so
\[J(n)\ge\sum_{k=1}^{n-1}\frac1{2(k+1)}=\frac{H_n-1}2.\]
\hint{3} For $n\ge2$: $H_n-1=\sum_{j=2}^n\frac1j\ge\int_2^{n+1}\frac{dt}t=\ln\frac{n+1}2\ge\frac12\ln n$ (because $\frac{n+1}2\ge\sqrt n$), and $1+\ln\frac\pi2<1.5<3\ln2\le3\ln n$. Hence $\frac14\ln n\le J(n)\le4\ln n$, and
\[\frac14\ln n\le I(n)\le2\pi\ln n\qquad(n\ge2).\]
(More precisely, $I(n)=\frac12\ln n+O(1)$.)

\problem{8}
\hint{1} Invert: $\dfrac1{a_{n+1}}=\dfrac1{a_n}+1+a_n$.
\hint{2} Hence $\frac1{a_n}\ge\frac1{a_0}+n$, so $a_n\to0$; then $\frac1{a_{n+1}}-\frac1{a_n}\to1$ and Stolz gives $\lim na_n=1$.

\problem{9}
\hint{1} Write $1+x+\dots+x^n=\frac{1-x^{n+1}}{1-x}$, so $\ln(1+\dots+x^n)=\ln(1-x^{n+1})-\ln(1-x)$. Since $\int_0^1\ln^2(1-x)\,dx=2$, you get, with $m=n+1$,
\[2-x_n=\int_0^1\Bigl(-\ln(1-x^m)\Bigr)\Bigl(-\ln(1-x)\Bigr)dx\ge0,\]
and the integrand is $O\bigl(\ln^2(1-x)\bigr)$ near $1$, so everything is finite.
\hint{2} Expand $-\ln(1-x^m)=\sum_{k\ge1}\frac{x^{mk}}k$ and use $\int_0^1x^N\bigl(-\ln(1-x)\bigr)dx=\frac{H_{N+1}}{N+1}$ (expand $-\ln(1-x)$ as well). This gives
\[2-x_n=\sum_{k\ge1}\frac{H_{mk+1}}{k\,(mk+1)}.\]
\hint{3} With $H_N=\ln N+O(1)$: $2-x_n=\frac1m\sum_{k\ge1}\frac{\ln m+\ln k+O(1)}{k^2}=\zeta(2)\frac{\ln m}m+O\bigl(\frac1m\bigr)$. In particular $x_n\to2$, and $\lim\frac n{\ln n}(2-x_n)=\zeta(2)=\frac{\pi^2}6$.

\problem{10}
\hint{1} Write $\sqrt[n]{1+x^n}-1=e^u-1$ with $u=\frac{\ln(1+x^n)}n\in[0,\frac{\ln2}n]$. Since $u\le e^u-1\le u+u^2$ for small $u$ and $u\le\frac{x^n}n$, the quadratic part contributes at most $n^2\int_0^1\frac{x^{2n}}{n^2}dx=\frac1{2n+1}\to0$.
\hint{2} So it suffices to find $\lim n\int_0^1\ln(1+x^n)\,dx$. Substitute $x=t^{1/n}$: $n\int_0^1\ln(1+x^n)\,dx=\int_0^1\frac{\ln(1+t)}t\,t^{1/n}\,dt$.
\hint{3} As $n\to\infty$ this tends to $\int_0^1\frac{\ln(1+t)}t\,dt$ (the integrand increases to the limit). Expanding $\ln(1+t)$, the integral equals $\sum_{k\ge1}\frac{(-1)^{k+1}}{k^2}=\frac{\pi^2}{12}$.
