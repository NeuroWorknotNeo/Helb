\hintstitle{Differentiable functions -- 1}{Class work}

\problem{1}
\hint{1} Decompose into partial fractions before differentiating: $\dfrac{x}{x^2-1}=\dfrac12\Bigl(\dfrac1{x-1}+\dfrac1{x+1}\Bigr)$.
\hint{2} Use $\bigl(\frac1{x-a}\bigr)^{(n)}=\frac{(-1)^n n!}{(x-a)^{n+1}}$. The answer is
$f^{(n)}(x)=\dfrac{(-1)^n n!}{2}\Bigl(\dfrac1{(x-1)^{n+1}}+\dfrac1{(x+1)^{n+1}}\Bigr)$.

\problem{2}
\hint{1} Since $f^{(k)}(0)=k!\,a_k$, where $f=\sum a_kx^k$ near $0$, you have to show that all Taylor coefficients of $f$ are positive. Products are awkward, so take logarithms.
\hint{2} $\ln f(x)=-\sum_i\ln(1-c_ix)=\sum_{k\ge1}\dfrac{s_k}{k}x^k$ with $s_k=c_1^k+\dots+c_n^k>0$. So $f=e^{g}$, where $g$ has zero constant term and positive coefficients.
\hint{3} From $f'=g'f$, compare coefficients: $a_0=1$ and $n a_n=\sum_{k=1}^{n}s_k\,a_{n-k}$. Induction on $n$ gives $a_n>0$.

\problem{3}
\hint{1} The hypothesis gives information about $(f^2)'=2ff'$. Suppose that $f(x)\to L$ and look for a contradiction.
\hint{2} Integrate $(f^2)'\ge 2\cos x$ over an interval where $\cos x\ge0$, e.g.\ $[2k\pi-\frac\pi2,\,2k\pi+\frac\pi2]$. This gives $f^2(2k\pi+\frac\pi2)-f^2(2k\pi-\frac\pi2)\ge4$ for every $k$.
\hint{3} If $f\to L$, then $f^2\to L^2$, so these differences must tend to $0$. Contradiction. (Boundedness of $f$ rules out an infinite limit, so only finite $L$ has to be considered.)

\problem{4}
\hint{1} This is the trick of Theorem 2 with a different multiplier. Find $\varphi>0$ such that $(\varphi f)'=x\varphi\,\bigl(f+\frac{f'}{x}\bigr)$.
\hint{2} $\varphi=e^{x^2/2}$ works. Write $f=\dfrac{e^{x^2/2}f}{e^{x^2/2}}$ and apply l'H\^opital (the denominator tends to $\infty$, which is all l'H\^opital needs).
\hint{3} The ratio of derivatives is $\dfrac{(f'+xf)e^{x^2/2}}{xe^{x^2/2}}=f+\dfrac{f'}{x}\to0$, so $f\to0$.

\problem{5}
\hint{1} Write $\frac{\sin x}{x}$ as an integral with a parameter: $\dfrac{\sin x}{x}=\displaystyle\int_0^1\cos(xt)\,dt$.
\hint{2} Differentiate under the integral sign: $f^{(n)}(x)=\displaystyle\int_0^1 t^n\cos\bigl(xt+\tfrac{n\pi}{2}\bigr)\,dt$.
\hint{3} Bound $|\cos|\le1$ to get $|f^{(n)}(x)|\le\int_0^1t^n\,dt=\frac1{n+1}$. The inequality is strict because the continuous integrand $|\cos(xt+n\pi/2)|$ is not identically $1$ on $[0,1]$ for $x>0$.
